scripted_gui = {
    capie_v1_ed__iron__bridge = {
        context_type = player_context
        window_name = "capie_v1_ed__iron__bridge"
        visible = { has_country_flag = iron_bridge_open }
        ai_enabled = { always = no }
        effects = {
            capie_v1_ed__iron__bridge_apply_click = {
                set_temp_variable = { iron_expected_epoch = iron_bridge_epoch }
                if = {
                    limit = {
                        check_variable = { temp_var:capie_cap > 1 }
                        check_variable = { temp_var:capie_epoch = temp_var:iron_expected_epoch }
                        check_variable = { temp_var:capie_nf = 2 }
                        check_variable = { temp_var:capie_f1_err = 0 }
                        check_variable = { temp_var:capie_f2_err = 0 }
                        check_variable = { temp_var:capie_f1_len > 0 }
                        check_variable = { temp_var:capie_f2_len > 0 }
                    }
                    set_variable = { iron_bridge_api = temp_var:capie_cap }
                    set_variable = { iron_bridge_seq_len = temp_var:capie_f1_len }
                    set_variable = { iron_bridge_seq_0 = temp_var:capie_f1_0 }
                    set_variable = { iron_bridge_arg_len = temp_var:capie_f2_len }
                    set_variable = { iron_bridge_arg_0 = temp_var:capie_f2_0 }
                    set_variable = { iron_bridge_arg_1 = temp_var:capie_f2_1 }
                    log = "IRON_CAPIE_ACK country=[Root.GetTag] epoch=[?iron_bridge_epoch] api=[?iron_bridge_api] seq_len=[?iron_bridge_seq_len] seq0=[?iron_bridge_seq_0] arg_len=[?iron_bridge_arg_len] arg0=[?iron_bridge_arg_0] arg1=[?iron_bridge_arg_1]"
                    add_to_variable = { iron_bridge_epoch = 1 }
                    clr_country_flag = iron_bridge_open
                }
                else = {
                    log = "IRON_CAPIE_REJECT country=[Root.GetTag] api=[?temp_var:capie_cap] epoch=[?temp_var:capie_epoch]"
                }
            }
            iron_bridge_close_click = { clr_country_flag = iron_bridge_open }
        }
        triggers = {
            capie_v1_ep__iron__bridge_visible = { always = no }
        }
    }
}
